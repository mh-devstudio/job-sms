async function sendDataToGoogleSheet() { let officeName = document.getElementById('officeInput').value.trim(); let userId = document.getElementById('userIdInput').value.trim(); let postCount = document.getElementById('postInput').value.trim(); let linkUrl = document.getElementById('linkInput').value.trim(); let smsInfo = document.getElementById('smsInput').value.trim(); let examStatus = document.getElementById('examStatusInput').value; if (officeName === "") { alert("দয়া করে অন্তত অফিস নাম লিখুন!"); return; } let formData = { officeName: officeName, userId: userId || "প্রযোজ্য নয়", postCount: postCount || "প্রযোজ্য নয়", linkUrl: linkUrl || "", smsInfo: smsInfo || "প্রযোজ্য নয়", examStatus: examStatus }; try { alert("ডেটা পাঠানো হচ্ছে, অনুগ্রহ করে অপেক্ষা করুন..."); await fetch(WEB_APP_URL, { method: "POST", mode: 'no-cors', // CORS ত্রুটি এড়ানোর জন্য এটি জরুরি headers: { "Content-Type": "application/json" }, body: JSON.stringify(formData) }); // যেহেতু no-cors মোডে সরাসরি রিমোট রেসপন্স পড়া যায় না, তাই ধরে নেওয়া হচ্ছে সফল হয়েছে alert("সফলভাবে গুগল শিটে সেভ হয়েছে!"); localCacheList.push(formData); localStorage.setItem('sheetCache', JSON.stringify(localCacheList)); displayResults(localCacheList); resetForm(); } catch (error) { console.error("ত্রুটি:", error); alert("সার্ভারে সংযোগ স্থাপন করা যায়নি। ইন্টারনেট কানেকশন চেক করুন।"); } }
